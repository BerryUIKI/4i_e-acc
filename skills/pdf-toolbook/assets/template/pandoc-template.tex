% ── Document class (ctexbook = Chinese-enabled book) ──
\documentclass[a4paper,12pt,oneside,fontset=none]{ctexbook}

% ── CJK typography (Source Han Serif body + Sans headings) ──
% ctex with fontset=none requires ALL CJK families to be defined.
% \setCJKmainfont only sets \CJKfamily{rm}, but ctex uses zh song/hei/kai/fs
% for body text — so we must configure every family via ctexset.
\usepackage{fontspec}
\setCJKmainfont{Source Han Serif SC}[
    BoldFont={Source Han Serif SC Bold}
]
\setCJKsansfont{Source Han Sans SC}[
    BoldFont={Source Han Sans SC Bold}
]
\setCJKmonofont{Source Han Sans SC}

\ctexset{
    zhsong = {Source Han Serif SC},
    zhhei  = {Source Han Sans SC},
    zhfs   = {Source Han Serif SC},    % 仿宋→宋体 fallback
    zhkai  = {Source Han Serif SC},    % 楷体→宋体 fallback
}

% ── Page geometry ──
\usepackage[top=2.5cm, bottom=2.5cm, left=2.8cm, right=2.3cm,
            headheight=23pt, headsep=8mm, footskip=12mm,
            includeheadfoot]{geometry}

% ── DollarHua brand colors ──
\usepackage{xcolor}
\definecolor{dh-pink}{HTML}{FEC6CD}
\definecolor{dh-blossom}{HTML}{F3A6AF}
\definecolor{dh-gold}{HTML}{E3A04B}
\definecolor{dh-navy}{HTML}{1D1E50}
\definecolor{dh-bronze}{HTML}{B86C40}
\definecolor{dh-skin}{HTML}{F9DECE}
\definecolor{dh-white}{HTML}{FCFDFD}
\definecolor{dh-red}{HTML}{E85D61}

% ── Unicode character support ──

\newunicodechar{×}{\ensuremath{\times}}

\newunicodechar{–}{\textendash}

\newunicodechar{°}{\ensuremath{^{\circ}}}

% ── Page style (headers / footers) ──
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.5pt}
\renewcommand{\headrule}{\hbox to\headwidth{%
  \color{dh-blossom}\leaders\hrule height \headrulewidth\hfill}}
\fancyhead[L]{\footnotesize\color{dh-navy}\leftmark}
\fancyhead[R]{\footnotesize\color{dh-navy}\mytitle}
\fancyfoot[C]{\thepage}
\fancypagestyle{plain}{
  \fancyhf{}
  \fancyfoot[C]{\thepage}
  \renewcommand{\headrulewidth}{0pt}
}

% ── First-line indent: 2 CJK characters ──
\setlength{\parindent}{2em}

% ── Chapter/section title styling ──
\usepackage{titlesec}

% Chapter: Bond Navy large text + pink decor rules
\titleformat{\chapter}[display]
  {\centering\Huge\bfseries\sffamily\color{dh-navy}}
  {\color{dh-blossom}\rule{0.6\textwidth}{2pt}\vspace{0.5cm}\\%
   \chaptertitlename\ \thechapter}
  {1em}
  {}
  [\vspace{0.3cm}\color{dh-blossom}\rule{0.3\textwidth}{1pt}]

% Section
\titleformat{\section}
  {\Large\bfseries\sffamily\color{dh-navy}}
  {\color{dh-blossom}\thesection}
  {1em}
  {}
  [\vspace{-0.3cm}{\color{dh-blossom}\titlerule[1pt]}]

% Subsection
\titleformat{\subsection}
  {\large\bfseries\sffamily\color{dh-navy}}
  {\color{dh-blossom}\thesubsection}
  {1em}
  {}

    \vspace{1cm}
    {\color{dh-blossom}\rule{0.3\textwidth}{1pt}\par}

    {\large\color{dh-blossom}$date$\par}
  \end{titlepage}
