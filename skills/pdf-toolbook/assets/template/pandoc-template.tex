% Chapter: display style with a band on top, the "第N章" label, the
% chapter title, and a trailing rule.
%
% titlesec's display-style "label" slot is sensitive: it must be a
% single, balanced group (no bare \par, no unprotected \vspace).
% We therefore wrap the entire label in a single {…} and use
% \vskip / \hrulefill instead of \par / \rule.
\titleformat{\chapter}[display]
  {\centering\Huge\bfseries\sffamily\color{dh-navy}}
  {{\color{dh-blossom}\hrule height 1.5pt}\vskip 0.7cm
   {\color{dh-navy-soft}\Large 第\ \thechapter\ 章}}
  {1.0em}
  {}
  [\vskip 0.6cm{\color{dh-blossom}\hrule height 1pt}]

\titlespacing*{\chapter}{0pt}{50pt}{40pt}

% Section: large bold with blossom number prefix (each section starts on a new page)
\newcommand{\sectionbreak}{\clearpage}
\titleformat{\section}
  {\Large\bfseries\sffamily\color{dh-navy}}
  {\color{dh-blossom}\thesection}
  {0.8em}
  {}
  [\vspace{-0.2em}{\color{dh-blossom}\titlerule[0.8pt]}]

\titlespacing*{\section}{0pt}{2.0em}{0.8em}

% Subsection: medium bold
\titleformat{\subsection}
  {\large\bfseries\sffamily\color{dh-navy}}
  {\color{dh-blossom}\thesubsection}
  {0.6em}
  {}

\titlespacing*{\subsection}{0pt}{1.6em}{0.6em}

% Subsubsection: small bold
\titleformat{\subsubsection}
  {\normalsize\bfseries\sffamily\color{dh-navy}}
  {\color{dh-blossom}\thesubsubsection}
  {0.5em}
  {}

% Subsection / subsubsection numbering: drop the \thesection component
% so a chapter with no parent \section does not render as "1.0.1, 1.0.2".
% We use plain chapter.X (and chapter.X.Y) instead.
\renewcommand{\thesubsection}{\thechapter.\arabic{subsection}}
\renewcommand{\thesubsubsection}{\thechapter.\arabic{subsection}.\arabic{subsubsection}}
